DROP DATABASE IF EXISTS SB_AM;
CREATE DATABASE SB_AM;
USE SB_AM;

CREATE TABLE article(
	id INT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
	regDate DATETIME NOT NULL,
	updateDate DATETIME NOT NULL,
	memberId INT UNSIGNED NOT NULL,
	boardId INT UNSIGNED NOT NULL,
	title VARCHAR(100) NOT NULL,
	`body` TEXT NOT NULL
);

ALTER TABLE article ADD COLUMN hitCnt INT UNSIGNED NOT NULL DEFAULT 0;

ALTER TABLE article ADD COLUMN equipment VARCHAR(20) NOT NULL DEFAULT '' COMMENT '장비 명' AFTER title;
ALTER TABLE article ADD COLUMN workStage VARCHAR(20) NOT NULL DEFAULT '' COMMENT '작업 단계' AFTER equipment;
ALTER TABLE article ADD COLUMN causeType VARCHAR(20) NOT NULL DEFAULT '원인 불명' COMMENT '원인 구분' AFTER workStage;
ALTER TABLE article ADD COLUMN `status` VARCHAR(20) NOT NULL DEFAULT '진행 중' COMMENT '진행 중 / 협의 중 / 해결 / 보류' AFTER causeType;
ALTER TABLE article ADD COLUMN resolverId INT UNSIGNED COMMENT '해결한 사람' AFTER `status`;

CREATE TABLE `member`(
	id INT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
	regDate DATETIME NOT NULL,
	updateDate DATETIME NOT NULL,
	loginId VARCHAR(20) NOT NULL UNIQUE,
	loginPw VARCHAR(100) NOT NULL COMMENT 'BCrypt 해시',
	authLevel INT UNSIGNED NOT NULL DEFAULT 1 COMMENT '권한 레벨 (일반 = 1, 관리자 = 0)',
	`name` VARCHAR(20) NOT NULL,
	nickname VARCHAR(20) NOT NULL,
	cellphoneNum VARCHAR(20) NOT NULL,
	email VARCHAR(50) NOT NULL,
	delStatus INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '탈퇴 여부 (탈퇴 전 = 0, 탈퇴 후 = 1)',
	delDate DATETIME COMMENT '탈퇴 날짜'
);

CREATE TABLE board(
	id INT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
	regDate DATETIME NOT NULL,
	updateDate DATETIME NOT NULL,
	`code` VARCHAR(20) NOT NULL COMMENT '사이트 식별용 영문 코드(siteA, siteB, suggestion)',
	`name` VARCHAR(20) NOT NULL COMMENT '게시판 이름'
);

ALTER TABLE board ADD COLUMN region VARCHAR(10) NOT NULL DEFAULT '국내' COMMENT '국내/해외/기타' AFTER `name`;

CREATE TABLE recommendPoint(
	id INT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
	memberId INT UNSIGNED NOT NULL,
	relTypeCode VARCHAR(20) NOT NULL COMMENT '관련 데이터 타입 코드',
	relId INT UNSIGNED NOT NULL COMMENT '관련 데이터 번호',
	`point` INT UNSIGNED NOT NULL DEFAULT 0
);

CREATE TABLE reply(
	id INT UNSIGNED NOT NULL PRIMARY KEY AUTO_INCREMENT,
	regDate DATETIME NOT NULL,
	updateDate DATETIME NOT NULL,
	memberId INT UNSIGNED NOT NULL,
	relTypeCode VARCHAR(20) NOT NULL COMMENT '관련 데이터 타입 코드',
	relId INT UNSIGNED NOT NULL COMMENT '관련 데이터 번호',
	`body` TEXT NOT NULL
);
